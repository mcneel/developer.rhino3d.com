{{- /* feature-card.html — a card for a page on this site (WWW-3694). It looks the page up and reads its thumbnail, title and short description, so a feature card picks up translated text and new artwork on its own; when the feature is new or changed in the current Rhino version it shows the "what's new" summary instead of the generic description. {{< card-gallery >}} {{< feature-card page="/features/shrinkwrap/index.md" >}} {{< /card-gallery >}} Required arguments: page path to the page, e.g. "/features/shrinkwrap/index.md" Optional arguments (each overrides what the page says): url link target (default: the page's own URL) title card title (default: the page's title) description card text (default: the what's-new summary for the current Rhino version, else short_description) image thumbnail: a path under /images/, or a file name looked up first in the images/ folder next to the page holding the card, then in the target page's bundle (default: the page's `thumbnail`) The box is not this shortcode's business: it builds the body — picture, linked title, description — and hands it to partials/card-container.html, the same partial the {{< card-container >}} shortcode renders through, so a feature card is the same card as every other card on the site. Styles: assets/scss/_shortcodes/_card.scss. Sits in a {{< card-gallery >}} grid. */ -}} {{- $error := "" -}} {{- if not .IsNamedParams -}} {{- $error = "Not using named params." -}} {{- else if not (.Get "page") -}} {{- $error = "page is a required parameter" -}} {{- end -}} {{- if $error -}} Error: {{ $error }}
Shortcode usage: {{< feature-card page=[path-to-page] >}} {{- else -}} {{- with .Site.GetPage (.Get "page") -}} {{- $page := . -}} {{- $latest := int $.Site.Params.latestRhinoVersion -}} {{- $link := $.Get "url" | default .RelPermalink -}} {{- $title := $.Get "title" | default .Title -}} {{- /* Description: the full short_description while the feature is brand new in this version, otherwise a 75-character teaser — of the what's-new summary if the page has one for this version, else of the generic short description. */ -}} {{- $description := "" -}} {{- if eq $latest (int .Params.included_in.since) -}} {{- $description = .Params.short_description -}} {{- else -}} {{- $summary := "" -}} {{- with .Params.whats_new -}} {{- with index . (printf "in_v%d_summary" $latest) -}} {{- $summary = . -}} {{- end -}} {{- end -}} {{- $description = truncate 75 (cond (ne $summary "") $summary .Params.short_description) -}} {{- end -}} {{- with $.Get "description" }}{{ $description = . }}{{ end -}} {{- /* Thumbnail, cropped to the media slot's 16:9. Either a file in the global /images/ folder or a bundle resource: an `image` override is looked for first in the images/ folder of the page holding the card, so card artwork can live next to the cards, then in the target page's bundle. An SVG is used as is (it cannot be resized), so draw it at 16:9. */ -}} {{- $thumbnail := "" -}} {{- with $.Get "image" | default .Params.thumbnail -}} {{- $image := "" -}} {{- if hasPrefix . "/images/" -}} {{- $image = resources.Get . -}} {{- else -}} {{- if $.Get "image" -}} {{- $image = $.Page.Resources.GetMatch (printf "images/%s" .) -}} {{- end -}} {{- if not $image -}} {{- $image = $page.Resources.GetMatch . -}} {{- end -}} {{- end -}} {{- with $image }}{{ if eq (lower (path.Ext .Name)) ".svg" }}{{ $thumbnail = . }}{{ else }}{{ $thumbnail = .Fill "800x450 Center" }}{{ end }}{{ end -}} {{- end -}} {{- /* Picture, linked title, description — the body an author would write by hand in {{< card-container >}}; the box partial turns that position into the media, title and text slots. */ -}} {{- partial "card-container.html" (dict "inner" (partial "card-page-body.html" (dict "link" $link "title" $title "description" $description "thumbnail" $thumbnail)) "class" "rh-card-tile") -}} {{- end -}} {{- end -}}
